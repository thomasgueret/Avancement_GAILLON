'===============================================================================================
'  REMPLISSAGE DES LISTES
'===============================================================================================

' Liste des numéros de fiche (sans doublon)
Private Sub ChargerFiches()

    Dim ws As Worksheet
    Dim lig As Long, derLig As Long
    Dim num As String, precedent As String

    Set ws = FeuilleSuivi()
    derLig = DerniereLigneFiches()

    cboFiche.Clear
    For lig = PREM_LIGNE To derLig
        num = TexteCellule(ws.Cells(lig, COL_FICHE))
        If num <> "" And num <> precedent Then
            If ListeContient(cboFiche, num) = False Then cboFiche.AddItem num
            precedent = num
        End If
    Next lig

    If cboFiche.ListCount > 0 Then cboFiche.ListIndex = 0

End Sub

' Liste des indices disponibles pour la fiche sélectionnée
Private Sub ChargerIndices()

    Dim ws As Worksheet
    Dim lig As Long, derLig As Long
    Dim num As String, ind As String
    Dim memoChargement As Boolean

    memoChargement = chargement
    chargement = True

    Set ws = FeuilleSuivi()
    derLig = DerniereLigneFiches()
    num = CStr(cboFiche.Value)

    nbIndices = 0
    ReDim lignesIndice(1 To 1000)
    cboIndice.Clear

    If num <> "" Then
        For lig = PREM_LIGNE To derLig
            If TexteCellule(ws.Cells(lig, COL_FICHE)) = num Then
                ind = TexteCellule(ws.Cells(lig, COL_INDICE))
                nbIndices = nbIndices + 1
                lignesIndice(nbIndices) = lig
                cboIndice.AddItem ind
            End If
        Next lig
    End If

    ' le dernier indice est en général celui que l'on veut viser
    If cboIndice.ListCount > 0 Then cboIndice.ListIndex = cboIndice.ListCount - 1

    chargement = memoChargement
    AfficherFiche

End Sub

' Liste des entités destinataires (en-têtes AB4, AE4, ...)
Private Sub ChargerEntites()

    Dim noms() As String
    Dim i As Long

    ListerEntites colsEntite, noms, nbEntites

    cboEntite.Clear
    For i = 1 To nbEntites
        cboEntite.AddItem noms(i)
    Next i

    If cboEntite.ListCount > 0 Then cboEntite.ListIndex = 0

End Sub

' Liste des visas possibles (onglet Données)
Private Sub ChargerVisas()

    Dim libelles() As String
    Dim i As Long

    ListerVisas codesVisa, libelles, nbVisas

    cboVisa.Clear
    For i = 1 To nbVisas
        cboVisa.AddItem libelles(i)
    Next i

End Sub

Private Function ListeContient(ByRef cbo As MSForms.ComboBox, ByVal valeur As String) As Boolean
    Dim i As Long
    For i = 0 To cbo.ListCount - 1
        If cbo.List(i) = valeur Then
            ListeContient = True
            Exit Function
        End If
    Next i
End Function

'===============================================================================================
'  AFFICHAGE DE LA FICHE SELECTIONNEE
'===============================================================================================
Private Sub AfficherFiche()

    Dim ws As Worksheet
    Dim lig As Long, colBloc As Long
    Dim visaExistant As String, dateExistante As String, texteExistant As String
    Dim i As Long
    Dim memoChargement As Boolean

    memoChargement = chargement
    chargement = True

    Set ws = FeuilleSuivi()
    lig = LigneSelectionnee()
    colBloc = ColonneEntiteSelectionnee()

    ' Rappel du sujet de la fiche
    If lig > 0 Then
        lblSujet.Caption = "Sujet : " & TexteCellule(ws.Cells(lig, COL_SUJET)) & _
                           "   (ligne " & lig & " du tableau)"
    Else
        lblSujet.Caption = "Sujet :"
    End If

    lblEtat.Caption = ""

    If lig > 0 And colBloc > 0 Then

        visaExistant = TexteCellule(ws.Cells(lig, colBloc))
        dateExistante = TexteCellule(ws.Cells(lig, colBloc + 1))
        texteExistant = ""
        If Not IsError(ws.Cells(lig, colBloc + 2).Value) Then
            texteExistant = CStr(ws.Cells(lig, colBloc + 2).Value)
        End If

        ' Pré-sélection du visa déjà saisi
        cboVisa.ListIndex = -1
        If visaExistant <> "" Then
            For i = 1 To nbVisas
                If UCase$(codesVisa(i)) = UCase$(visaExistant) Then
                    cboVisa.ListIndex = i - 1
                    Exit For
                End If
            Next i
        End If

        If dateExistante <> "" Then
            txtDate.Text = dateExistante
        Else
            txtDate.Text = Format$(Date, "dd/mm/yyyy")
        End If

        ' On ne remplace le texte saisi que si la cellule contient déjà quelque chose
        If Len(texteExistant) > 0 Then
            txtReponse.Text = texteExistant
        ElseIf Len(txtReponse.Text) = 0 Then
            txtReponse.Text = ""
        End If

        If visaExistant <> "" Or Len(texteExistant) > 0 Then
            lblEtat.Caption = "Attention : une réponse existe déjà pour cette entité " & _
                              "(visa " & visaExistant & "). L'enregistrement la remplacera."
        End If

    End If

    chargement = memoChargement

End Sub

Private Function LigneSelectionnee() As Long
    If cboIndice.ListIndex >= 0 And cboIndice.ListIndex < nbIndices Then
        LigneSelectionnee = lignesIndice(cboIndice.ListIndex + 1)
    End If
End Function

Private Function ColonneEntiteSelectionnee() As Long
    If cboEntite.ListIndex >= 0 And cboEntite.ListIndex < nbEntites Then
        ColonneEntiteSelectionnee = colsEntite(cboEntite.ListIndex + 1)
    End If
End Function

'===============================================================================================
'  EVENEMENTS
'===============================================================================================
Private Sub cboFiche_Change()
    If chargement Then Exit Sub
    ChargerIndices
End Sub

Private Sub cboIndice_Change()
    If chargement Then Exit Sub
    AfficherFiche
End Sub

Private Sub cboEntite_Change()
    If chargement Then Exit Sub
    AfficherFiche
End Sub

Private Sub btnAnnuler_Click()
    Me.Hide
End Sub

Private Sub btnOK_Click()

    Dim ws As Worksheet
    Dim lig As Long, colBloc As Long
    Dim codeVisa As String
    Dim dateVisa As Variant
    Dim recap As String

    Set ws = FeuilleSuivi()

    lig = LigneSelectionnee()
    colBloc = ColonneEntiteSelectionnee()

    If lig = 0 Then
        MsgBox "Sélectionnez le numéro de fiche et son indice.", vbExclamation, TITRE
        Exit Sub
    End If

    If colBloc = 0 Then
        MsgBox "Sélectionnez l'entité qui répond à la fiche.", vbExclamation, TITRE
        cboEntite.SetFocus
        Exit Sub
    End If

    If cboVisa.ListIndex < 0 Then
        MsgBox "Sélectionnez le visa (F, FAO, S, R, ATT, EXP).", vbExclamation, TITRE
        cboVisa.SetFocus
        Exit Sub
    End If
    codeVisa = codesVisa(cboVisa.ListIndex + 1)

    If Trim$(txtDate.Text) = "" Then
        dateVisa = Empty
    ElseIf IsDate(txtDate.Text) Then
        dateVisa = CDate(txtDate.Text)
    Else
        MsgBox "La date saisie n'est pas valide (format attendu : jj/mm/aaaa).", vbExclamation, TITRE
        txtDate.SetFocus
        Exit Sub
    End If

    recap = "Fiche n° " & cboFiche.Value & " - indice " & cboIndice.Value & vbLf & _
            "Entité : " & cboEntite.Value & vbLf & _
            "Visa : " & codeVisa & vbLf & _
            "Date : " & IIf(IsDate(dateVisa), Format$(dateVisa, "dd/mm/yyyy"), "(vide)") & vbLf & _
            "Cellules : " & ws.Cells(lig, colBloc).Address(False, False) & " à " & _
            ws.Cells(lig, colBloc + 2).Address(False, False) & vbLf & vbLf & _
            "Confirmez-vous l'enregistrement ?"

    If MsgBox(recap, vbOKCancel + vbQuestion, TITRE) <> vbOK Then Exit Sub

    If Not EnregistrerReponse(lig, colBloc, codeVisa, dateVisa, txtReponse.Text) Then Exit Sub

    If MsgBox("Réponse enregistrée." & vbLf & vbLf & _
              "Voulez-vous saisir une autre réponse ?", vbYesNo + vbInformation, TITRE) = vbYes Then
        chargement = True
        txtReponse.Text = ""
        txtDate.Text = Format$(Date, "dd/mm/yyyy")
        chargement = False
        AfficherFiche
        cboEntite.SetFocus
    Else
        Me.Hide
    End If

End Sub
